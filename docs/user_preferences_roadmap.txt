# User Preferences Implementation Roadmap

**Project:** Transportation Portal  
**Feature:** Server-Side User Preferences with Authentication  
**Goal:** User preferences persist across devices, browsers, and cache clears  
**Created:** October 9, 2025

---

## Executive Summary

This roadmap outlines the implementation of a complete user authentication and preferences system that ensures user-defined settings persist across:
- Different devices/computers
- Different browsers
- After browser cache/data wipes
- After logout/login cycles

**Core Principle:** User preferences stored on server (Elasticsearch), retrieved on login, applied to UI.

---

## Prerequisites

### Already Implemented ✅
- Elasticsearch cluster operational (`data` namespace)
- Backend API framework (Node.js + Express)
- Frontend state management (StateManagementService)
- JWT dependencies installed (jsonwebtoken, bcryptjs)

### Technology Stack
- **Auth:** JWT tokens (httpOnly cookies or Authorization header)
- **Storage:** Elasticsearch index `user-preferences`
- **Backend:** Node.js + Express + JWT + bcryptjs
- **Frontend:** Angular 13 + RxJS

**Tasks:**
1. Create `backend/middleware/authMiddleware.js`:
```javascript
const jwt = require('jsonwebtoken');

module.exports = (req, res, next) => {
  // Extract token from Authorization header or cookie
  const token = req.cookies.token || 
                req.headers.authorization?.replace('Bearer ', '');
  
  if (!token) {
    return res.status(401).json({ error: 'Authentication required' });
  }
  
  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    req.user = decoded; // Attach user info to request
    next();
  } catch (error) {
    return res.status(401).json({ error: 'Invalid or expired token' });
  }
};
```

**Estimated Time:** 3-4 hours

**Tasks:**
1. Create `frontend/src/app/core/services/auth.service.ts`:
```typescript
@Injectable({ providedIn: 'root' })
export class AuthService {
  private currentUserSubject = new BehaviorSubject<User | null>(null);
  public currentUser$ = this.currentUserSubject.asObservable();
  
  private tokenKey = 'auth_token';
  
  constructor(private http: HttpClient) {
    this.checkStoredToken();
  }
  
  login(email: string, password: string): Observable<User> {
    return this.http.post<{user: User, token: string}>(
      `${environment.apiUrl}/auth/login`,
      { email, password }
    ).pipe(
      tap(response => {
        this.setToken(response.token);
        this.currentUserSubject.next(response.user);
      }),
      map(response => response.user)
    );
  }
  
  logout(): void {
    this.removeToken();
    this.currentUserSubject.next(null);
    // Optional: Call backend logout endpoint
  }
  
  register(email: string, password: string, displayName: string): Observable<void> {
    return this.http.post<void>(
      `${environment.apiUrl}/auth/register`,
      { email, password, displayName }
    );
  }
  
  getCurrentUser(): Observable<User> {
    return this.http.get<User>(`${environment.apiUrl}/auth/me`);
  }
  
  isAuthenticated(): boolean {
    return !!this.getToken() && !!this.currentUserSubject.value;
  }
  
  private setToken(token: string): void {
    localStorage.setItem(this.tokenKey, token);
  }
  
  private getToken(): string | null {
    return localStorage.getItem(this.tokenKey);
  }
  
  private removeToken(): void {
    localStorage.removeItem(this.tokenKey);
  }
  
  private checkStoredToken(): void {
    const token = this.getToken();
    if (token) {
      // Verify token is still valid
      this.getCurrentUser().subscribe({
        next: user => this.currentUserSubject.next(user),
        error: () => this.removeToken()
      });
    }
  }
}
```

2. Create `frontend/src/app/core/interceptors/auth.interceptor.ts`:
```typescript
@Injectable()
export class AuthInterceptor implements HttpInterceptor {
  constructor(private auth: AuthService) {}
  
  intercept(req: HttpRequest<any>, next: HttpHandler): Observable<HttpEvent<any>> {
    const token = localStorage.getItem('auth_token');
    
    if (token) {
      req = req.clone({
        setHeaders: {
          Authorization: `Bearer ${token}`
        }
      });
    }
    
    return next.handle(req);
  }
}
```

3. Register interceptor in `app.module.ts`:
```typescript

  {
    provide: HTTP_INTERCEPTORS,
    useClass: AuthInterceptor,
    multi: true
  }
]
```

**Validation:**
- All HTTP requests should include Authorization header when logged in
- Token persists in localStorage across page refreshes

@Injectable({ providedIn: 'root' })
export class AuthGuard implements CanActivate {
  constructor(
    private auth: AuthService,
    private router: Router
  ) {}
  
  canActivate(route: ActivatedRouteSnapshot): boolean {
    if (this.auth.isAuthenticated()) {
      return true;
    }
    
    // Save intended URL for redirect after login
    const returnUrl = route.url.map(segment => segment.path).join('/');
    
    this.router.navigate(['/login'], {
      queryParams: { returnUrl }
    });
    
    return false;
  }
}
```

@Injectable({ providedIn: 'root' })
export class PreferencesService {
  private preferencesSubject = new BehaviorSubject<UserPreferences | null>(null);
  public preferences$ = this.preferencesSubject.asObservable();
  
  constructor(
    private http: HttpClient,
    private auth: AuthService
  ) {
    // Load preferences when user logs in
    this.auth.currentUser$.pipe(
      switchMap(user => {
        if (user) {
          return this.loadPreferences();
        } else {
          this.preferencesSubject.next(null);
          return of(null);
        }
      })
    ).subscribe();
  }
  
  loadPreferences(): Observable<UserPreferences> {
    return this.http.get<UserPreferences>(
      `${environment.apiUrl}/preferences/me`
    ).pipe(
      tap(prefs => this.preferencesSubject.next(prefs))
    );
  }
  
  updatePreferences(updates: Partial<UserPreferences>): Observable<UserPreferences> {
    const current = this.preferencesSubject.value;
    const updated = { ...current, ...updates };
    
    return this.http.put<UserPreferences>(
      `${environment.apiUrl}/preferences/me`,
      updated
    ).pipe(
      tap(prefs => this.preferencesSubject.next(prefs))
    );
  }
  
  saveSearch(name: string, filters: SearchFilters): Observable<void> {
    return this.http.post<void>(
      `${environment.apiUrl}/preferences/me/saved-searches`,
      { name, filters }
    ).pipe(
      switchMap(() => this.loadPreferences()),
      map(() => undefined)
    );
  }
  
  deleteSavedSearch(searchId: string): Observable<void> {
    return this.http.delete<void>(
      `${environment.apiUrl}/preferences/me/saved-searches/${searchId}`
    ).pipe(
      switchMap(() => this.loadPreferences()),
      map(() => undefined)
    );
  }
}
```

2. Update `StateManagementService` to use preferences:
```typescript
constructor(
  private routeState: RouteStateService,
  private preferences: PreferencesService
) {
  // Initialize state with preferences
  combineLatest([
    this.routeState.queryParams$,
    this.preferences.preferences$
  ]).pipe(
    map(([params, prefs]) => {
      const urlFilters = this.routeState.paramsToFilters(params);
      const defaultFilters = {
        page: 1,
        size: prefs?.defaultPageSize || 20,
        sort: prefs?.defaultSort?.field || 'year',
        sortOrder: prefs?.defaultSort?.order || 'desc'
      };
      
      // URL params override preferences
      return { ...defaultFilters, ...urlFilters };
    })
  ).subscribe(filters => {
    this.updateFilters(filters);
  });
}
```

```typescript
export function initializeApp(
  auth: AuthService,
  preferences: PreferencesService
): () => Promise<void> {
  return () => {
    return new Promise((resolve) => {
      // Check if user has stored token
      const token = localStorage.getItem('auth_token');
      
      if (token) {
        // Verify token and load user
        auth.getCurrentUser().pipe(
          switchMap(user => {
            // Load preferences after user verified
            return preferences.loadPreferences();
          })
        ).subscribe({
          next: () => resolve(),
          error: () => {
            // Token invalid, clear and continue
            auth.logout();
            resolve();
          }
        });
      } else {
        resolve();
      }
    });
  };
}
```
